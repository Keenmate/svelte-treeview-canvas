RENDER CONTEXT
--------------
interface CanvasRenderContext<T> {
  ctx: CanvasRenderingContext2D;
  node: LTreeNode<T>;
  label: string;
  badgeContent: string | null;
  bounds: CanvasNodeBounds;
  state: CanvasNodeState;
  lod: LodLevel;
  depthColor: string;
  config: CanvasVisualConfig;
  theme: CanvasTheme;
}

interface CanvasNodeBounds {
  x: number; y: number;
  w: number; h: number;
  cx: number; cy: number;
  depth: number;
}

interface CanvasNodeState {
  isSelected: boolean;
  isHovered: boolean;
  isDragSource: boolean;
  isDropTarget: boolean;
  isSearchMatch: boolean;
  isCurrentSearchResult: boolean;
  isSearchDimmed: boolean;
}

CALLBACK TYPES
--------------
type RenderNodeCallback<T> = (rctx: CanvasRenderContext<T>) => void;
type RenderSlotCallback<T> = (rctx: CanvasRenderContext<T>) => void;
type MeasureNodeWidthCallback<T> = (
  ctx: CanvasRenderingContext2D,
  node: LTreeNode<T>,
  config: CanvasVisualConfig
) => number;
type MeasureNodeHeightCallback<T> = (
  ctx: CanvasRenderingContext2D,
  node: LTreeNode<T>,
  config: CanvasVisualConfig
) => number;
type GetNodeLabelCallback<T> = (node: LTreeNode<T>) => string;
type GetBadgeContentCallback<T> = (node: LTreeNode<T>) => string | null;

FOCUS TYPES
-----------
type FocusAnchor =
  | 'center' | 'top-left' | 'top-center' | 'top-right'
  | 'center-left' | 'center-right'
  | 'bottom-left' | 'bottom-center' | 'bottom-right'
  | { x: number; y: number };

type FocusZoom = number | 'keep' | 'auto';

interface FocusOptions {
  anchor?: FocusAnchor;
  zoom?: FocusZoom;
  padding?: number;
  shouldAnimate?: boolean;   // default true
  shouldSelect?: boolean;    // default true (updates focusedPath)
}

  parent?: LayoutNode<T>;
}

CONTEXT MENU (from core)
-------------------------
// TreeMenuContext<T> = { node, path, parent, siblings, controller }
interface TreeMenuItem<T> {            // core MenuItem<TreeMenuContext<T>, TreeAction>
  id?: string;
  action?: TreeAction;                 // built-in behaviour (or just the name as an entry)
  text?: string;
  icon?: string;
  tooltip?: string;
  shortcut?: string;
  cssClass?: string | string[];
  isDisabled?: boolean;
  isVisible?: boolean;
  isDanger?: boolean;
  isToggled?: boolean;
  // getTextCallback / getIconCallback / getIsDisabledCallback / … (ctx) win over the static fields
  onClick?(detail: TreeActionEventDetail<T>): void | Promise<void>;
  children?: TreeMenuEntry<T>[];
  getChildrenCallback?(ctx): TreeMenuEntry<T>[] | Promise<TreeMenuEntry<T>[]>;
  shouldCloseOnClick?: boolean;        // default true
}

interface MenuDivider { divider: true; text?: string; }   // menuDivider(text?)

type TreeMenuEntry<T> = TreeMenuItem<T> | MenuDivider | TreeAction;

type TreeAction = 'copy' | 'cut' | 'paste' | 'delete' | 'expand' | 'collapse'
  | 'expand-subtree' | 'collapse-subtree' | 'expand-all' | 'collapse-all'
  | 'sort' | 'sort-subtree' | 'sort-all';
